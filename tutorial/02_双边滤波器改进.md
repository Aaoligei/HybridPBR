# 双边滤波器改进 - 添加G-Buffer引导

## 本章目标

修改现有的降噪着色器，添加深度和法线权重，实现几何感知的边缘保护。

## 3.1 改进思路总览

## 3.2 第一步：修改C++代码

### 3.2.1 修改Denoiser类接口

// 原接口：

// 新接口：
void Denoise(const std::shared_ptr<Texture>& input, 
            const std::shared_ptr<Texture>& output,

            const std::shared_ptr<Texture>& normalTex = nullptr,   // 新增
            const std::shared_ptr<Texture>& depthTex = nullptr);   // 新增

### 3.2.2 修改Denoiser.cpp实现

```cpp
// Denoiser.cpp 修改方案

void Denoiser::Denoise(const std::shared_ptr<Texture>& input, 
                       const std::shared_ptr<Texture>& output, 
                       float strength,
                       const std::shared_ptr<Texture>& normalTex,
                       const std::shared_ptr<Texture>& depthTex) {
    if (!denoiserShader || !input || !output) {
        return;
    }
    
    this->strength = strength;
    
    denoiserShader->Use();
    
    // 设置统一变量
    denoiserShader->SetInt("width", static_cast<int>(width));
    denoiserShader->SetInt("height", static_cast<int>(height));
    denoiserShader->SetFloat("strength", strength);
    
    // 绑定输入纹理
    input->Bind(0);
    denoiserShader->SetInt("inputTexture", 0);
    
    // 【新增】绑定G-Buffer纹理
    if (normalTex) {
        normalTex->Bind(1);
        denoiserShader->SetInt("normalTexture", 1);
        denoiserShader->SetInt("useNormal", 1);
    } else {
        denoiserShader->SetInt("useNormal", 0);
    }
    
    if (depthTex) {
        depthTex->Bind(2);
        denoiserShader->SetInt("depthTexture", 2);
        denoiserShader->SetInt("useDepth", 1);
    } else {
        denoiserShader->SetInt("useDepth", 0);
    }
    
    // 绑定输出纹理
    output->BindImage(0, 0, GL_WRITE_ONLY);
    
    // 分派计算着色器
    uint32_t groupsX = (width + 15) / 16;
    uint32_t groupsY = (height + 15) / 16;
    denoiserShader->Dispatch(groupsX, groupsY, 1);
    
    ComputeShader::ImageAccessBarrier();
    
    // 解绑
    glBindImageTexture(0, 0, 0, GL_FALSE, 0, GL_READ_WRITE, GL_RGBA32F);
    input->Unbind();
}
```

## 3.3 第二步：修改着色器代码

### 3.3.1 添加新的uniform和采样器

```glsl
// denoiser.comp 修改方案

#version 460 core
layout(local_size_x = 16, local_size_y = 16) in;

// 原有绑定
layout(binding = 0) uniform sampler2D inputTexture;
layout(binding = 0, rgba32f) uniform writeonly image2D outputImage;

// 【新增】G-Buffer纹理
layout(binding = 1) uniform sampler2D normalTexture;
layout(binding = 2) uniform sampler2D depthTexture;

uniform int width;
uniform int height;
uniform float strength;

// 【新增】开关
uniform int useNormal;
uniform int useDepth;

// 【新增】滤波参数
const float SIGMA_NORMAL = 0.1;   // 法线差异阈值
const float SIGMA_DEPTH = 0.01;   // 深度差异阈值（需要根据场景调整）
```

### 3.3.2 实现新的权重计算

```glsl
// 【新增】计算法线权重
float CalculateNormalWeight(vec3 n1, vec3 n2) {
    // 法线是单位向量，点积等于cos(夹角)
    // 夹角越大，权重越小
    float dotProduct = dot(n1, n2);
    
    // 法线应该非常接近才能混合
    // 如果点积接近1，说明法线相似
    // 如果点积接近0或负数，说明法线差异很大
    return pow(max(dotProduct, 0.0), 32.0);  // 32次幂让权重衰减更快
}

// 【新增】计算深度权重
float CalculateDepthWeight(float d1, float d2, float centerDepth) {
    // 深度差异相对于中心深度的比例
    float depthDiff = abs(d1 - d2);
    float relativeDiff = depthDiff / max(centerDepth, 0.001);
    
    // 指数衰减
    return exp(-relativeDiff * relativeDiff / (2.0 * SIGMA_DEPTH * SIGMA_DEPTH));
}
```

### 3.3.3 修改主滤波循环

```glsl
void main() {
    ivec2 pixelCoords = ivec2(gl_GlobalInvocationID.xy);
    if (pixelCoords.x >= width || pixelCoords.y >= height) return;

    // 获取中心像素数据
    vec3 centerColor = texelFetch(inputTexture, pixelCoords, 0).rgb;
    
    // 【新增】获取中心像素的法线和深度
    vec3 centerNormal = vec3(0.0, 1.0, 0.0);
    float centerDepth = 1.0;
    
    if (useNormal > 0) {
        centerNormal = texelFetch(normalTexture, pixelCoords, 0).rgb;
        // 如果法线存储在[0,1]范围，需要转换到[-1,1]
        centerNormal = normalize(centerNormal * 2.0 - 1.0);
    }
    
    if (useDepth > 0) {
        centerDepth = texelFetch(depthTexture, pixelCoords, 0).r;
    }
    
    // 颜色权重参数
    float sigmaColor = 0.1 + (strength * 1.0);

    vec3 sumColor = vec3(0.0);
    float sumWeight = 0.0;

    for (int y = -KERNEL_RADIUS; y <= KERNEL_RADIUS; y++) {
        for (int x = -KERNEL_RADIUS; x <= KERNEL_RADIUS; x++) {
            ivec2 sampleCoords = pixelCoords + ivec2(x, y);
            sampleCoords = clamp(sampleCoords, ivec2(0), ivec2(width - 1, height - 1));

            vec3 sampleColor = texelFetch(inputTexture, sampleCoords, 0).rgb;
            
            // 1. 空间权重
            float dist2 = float(x*x + y*y);
            float weightSpatial = exp(-dist2 / (2.0 * SIGMA_SPATIAL * SIGMA_SPATIAL));
            
            // 2. 颜色权重
            vec3 diff = sampleColor - centerColor;
            float colorDist2 = dot(diff, diff);
            float weightColor = exp(-colorDist2 / (2.0 * sigmaColor * sigmaColor));
            
            // 【新增】3. 法线权重
            float weightNormal = 1.0;
            if (useNormal > 0) {
                vec3 sampleNormal = texelFetch(normalTexture, sampleCoords, 0).rgb;
                sampleNormal = normalize(sampleNormal * 2.0 - 1.0);
                weightNormal = CalculateNormalWeight(centerNormal, sampleNormal);
            }
            
            // 【新增】4. 深度权重
            float weightDepth = 1.0;
            if (useDepth > 0) {
                float sampleDepth = texelFetch(depthTexture, sampleCoords, 0).r;
                weightDepth = CalculateDepthWeight(centerDepth, sampleDepth, centerDepth);
            }
            
            // 【修改】综合权重
            float weight = weightSpatial * weightColor * weightNormal * weightDepth;
            
            sumColor += sampleColor * weight;
            sumWeight += weight;
        }
    }

    vec3 finalColor = sumColor / max(sumWeight, 0.0001);
    imageStore(outputImage, pixelCoords, vec4(finalColor, 1.0));
}
```

## 3.4 参数调优指南

### 3.4.1 各参数的作用

```
┌────────────────────────────────────────────────────────────────┐
│  参数            │  作用                  │  调大效果          │
├────────────────────────────────────────────────────────────────┤
│  SIGMA_SPATIAL   │  空间滤波范围          │  更强的空间模糊    │
│  sigmaColor      │  颜色容差              │  颜色差异大的也能  │
│                  │                        │  混合（可能模糊边缘）│
│  SIGMA_NORMAL    │  法线容差              │  法线差异大的也能  │
│                  │                        │  混合（可能模糊几何边缘）│
│  SIGMA_DEPTH     │  深度容差              │  深度差异大的也能  │
│                  │                        │  混合（可能模糊前后边界）│
│  法线权重幂次    │  边缘敏感度            │  更强的边缘保护    │
│  (当前设为32)    │                        │  但可能保留噪声    │
└────────────────────────────────────────────────────────────────┘
```

### 3.4.2 不同场景的推荐参数

```glsl
// 场景1：静态室内场景，强调边缘保护
const float SIGMA_NORMAL = 0.05;   // 更严格
const float SIGMA_DEPTH = 0.005;   // 更严格

// 场景2：户外大场景，强调降噪效果

// 场景3：高噪声输入（样本少）

```

### 3.4.3 深度权重的特殊处理

深度权重需要考虑透视投影的特性：

```glsl
// 改进的深度权重计算
float CalculateDepthWeightImproved(float d1, float d2, vec2 pixelOffset) {
    // 深度差异相对于像素间距的深度梯度
    // 考虑透视效果：远处的深度变化应该容忍度更大
    
    float depthDiff = abs(d1 - d2);
    
    // 方法1：线性深度
    // 如果你的深度缓冲区存储的是线性深度，直接使用
    float relativeDiff = depthDiff / max(d1, 0.001);
    
    // 方法2：非线性深度（透视投影后）
    // 需要考虑深度缓冲区的非线性特性
    // 近处：深度变化敏感
    // 远处：深度变化不敏感
    float depthSensitivity = 0.1 / max(d1 * d1, 0.0001);
    relativeDiff = depthDiff * depthSensitivity;
    
    return exp(-relativeDiff * relativeDiff / (2.0 * SIGMA_DEPTH * SIGMA_DEPTH));
}
```

## 3.5 如何获取G-Buffer纹理

### 3.5.1 如果项目已有G-Buffer

查看`src/rendering/deferred/GBuffer.h`，你的项目已经实现了G-Buffer：

```cpp
// GBuffer.h 中定义的纹理类型
enum class GBufferTextureType {
    Position,
    Normal,
    Albedo,
    MetallicRoughnessAO,
    Emissive
};
```

你需要做的是：
1. 在光追渲染时，将G-Buffer的法线纹理传递给降噪器
2. 可能需要单独生成深度纹理

### 3.5.2 生成深度纹理

如果G-Buffer没有单独的深度纹理，可以从Position纹理提取：

```glsl
// compute shader: extract_depth.comp
#version 460 core
layout(local_size_x = 16, local_size_y = 16) in;

layout(binding = 0) uniform sampler2D positionTexture;
layout(binding = 0, r32f) uniform writeonly image2D depthImage;

uniform vec3 cameraPos;

void main() {
    ivec2 coords = ivec2(gl_GlobalInvocationID.xy);
    vec3 position = texelFetch(positionTexture, coords, 0).rgb;
    float depth = length(position - cameraPos);
    imageStore(depthImage, coords, vec4(depth, 0.0, 0.0, 1.0));
}
```

## 3.7 常见问题与解决

### 问题1：法线纹理格式不正确

**症状**：开启法线权重后，图像全黑或全白

**原因**：法线纹理存储格式与着色器预期不符

**解决**：
```glsl
// 检查法线纹理的实际存储方式
// 方式1：[-1,1] 存储为 [0,1]
vec3 normal = texelFetch(normalTexture, coords, 0).rgb;
normal = normal * 2.0 - 1.0;  // 转换到[-1,1]

// 方式2：已经是[-1,1]
vec3 normal = texelFetch(normalTexture, coords, 0).rgb;
// 直接使用

// 方式3：只存储XY，Z通过计算得到
vec2 normalXY = texelFetch(normalTexture, coords, 0).rg;
float normalZ = sqrt(1.0 - dot(normalXY, normalXY));
vec3 normal = vec3(normalXY, normalZ);
```

### 问题2：深度纹理精度问题

**症状**：深度权重导致远处的物体被过度模糊

**原因**：深度缓冲区精度不均匀（近处精度高，远处精度低）

**解决**：
```glsl
// 使用深度值的对数来计算权重
float logDepth1 = log(max(d1, 0.001));
float logDepth2 = log(max(d2, 0.001));
float depthDiff = abs(logDepth1 - logDepth2);
float weight = exp(-depthDiff * depthDiff / (2.0 * SIGMA_DEPTH));
```

### 问题3：性能下降明显

**症状**：帧率下降超过30%

**原因**：额外的纹理采样和计算

**解决**：
```glsl
// 优化1：减少纹理采样次数
// 先计算空间权重，如果太小就跳过
float weightSpatial = exp(-dist2 / (2.0 * SIGMA_SPATIAL * SIGMA_SPATIAL));
if (weightSpatial < 0.01) continue;  // 跳过贡献太小的样本

// 优化2：使用更小的滤波核
const int KERNEL_RADIUS = 3;  // 从4改为3

// 优化3：分离滤波（两遍处理）
// 第一遍：水平滤波
// 第二遍：垂直滤波
// 这样 O(n²) 变成 O(2n)
```
