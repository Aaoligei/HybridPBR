```cpp
// SobolSampler.cpp

#include "SobolSampler.h"

namespace HybridPBR {

std::vector<glm::vec4> SobolSampler::sobolData;
std::shared_ptr<Texture> SobolSampler::sobolTexture;

// Sobol方向数（这里只列出部分，完整版本需要更多）
static const uint32_t sobolV0[32] = {
    0x80000000u, 0xc0000000u, 0xa0000000u, 0xf0000000u,
    0x88000000u, 0xcc000000u, 0xaa000000u, 0xff000000u,
    // ... 完整的32个值
};

static const uint32_t sobolV1[32] = {
    0x80000000u, 0xc0000000u, 0x60000000u, 0x90000000u,
    0x8c000000u, 0xc6000000u, 0x69000000u, 0x9f000000u,
    // ... 完整的32个值
};

void SobolSampler::Initialize() {
    sobolData.resize(TABLE_SIZE);
    
    for (int i = 0; i < TABLE_SIZE; i++) {
        float x = static_cast<float>(Sobol(i, 0)) / 4294967296.0f;
        float y = static_cast<float>(Sobol(i, 1)) / 4294967296.0f;
        sobolData[i] = glm::vec4(x, y, 0.0f, 1.0f);
    }
    
    // 创建纹理
    sobolTexture = std::make_shared<Texture>();
    sobolTexture->Create2D(TABLE_SIZE, 1, GL_RGBA32F, GL_RGBA, GL_FLOAT, 
                           sobolData.data());
    sobolTexture->SetFilter(TextureFilter::NEAREST, TextureFilter::NEAREST);
    sobolTexture->SetWrapMode(TextureWrap::CLAMP_TO_EDGE, TextureWrap::CLAMP_TO_EDGE);
}

uint32_t SobolSampler::Sobol(uint32_t index, uint32_t dimension) {
    uint32_t result = 0u;
    uint32_t bit = 0u;
    
    while (index > 0u) {
        if ((index & 1u) != 0u) {
            if (dimension == 0u) {
                result ^= sobolV0[bit];
            } else {
                result ^= sobolV1[bit];
            }
        }
        index >>= 1u;
        bit++;
    }
    
    return result;
}

glm::vec2 SobolSampler::Get2D(uint32_t index, uint32_t dimension) {
    uint32_t i = index % TABLE_SIZE;
    return glm::vec2(sobolData[i].x, sobolData[i].y);
}

void SobolSampler::Shutdown() {
    sobolTexture = nullptr;
    sobolData.clear();
}

} // namespace HybridPBR
```

```cpp
// MotionVectorCalculator.cpp

#include "MotionVectorCalculator.h"
#include "utils/Logger.h"

namespace HybridPBR {

MotionVectorCalculator::MotionVectorCalculator() 
    : width(0), height(0) {
}

MotionVectorCalculator::~MotionVectorCalculator() {
}

bool MotionVectorCalculator::Initialize(uint32_t width, uint32_t height) {
    this->width = width;
    this->height = height;
    
    // 创建运动向量纹理
    motionVectorTexture = std::make_shared<Texture>();
    motionVectorTexture->Create2D(width, height, GL_RG16F, GL_RG, GL_FLOAT);
    motionVectorTexture->SetFilter(TextureFilter::NEAREST, TextureFilter::NEAREST);
    
    // 加载着色器
    motionShader = std::make_shared<ComputeShader>();
    if (!motionShader->LoadFromFile("assets/shaders/compute/motion_vectors.comp")) {
        LOG_ERROR("Failed to load motion vector shader");
        return false;
    }
    
    return true;
}

void MotionVectorCalculator::Calculate(
    const std::shared_ptr<Texture>& currentDepth,
    const std::shared_ptr<Texture>& output) {
    
    if (!motionShader) return;
    
    motionShader->Use();
    
    // 设置相机矩阵
    motionShader->SetMat4("currentViewProjectionInverse", currentViewProjectionInverse);
    motionShader->SetMat4("previousViewProjection", previousViewProjection);
    
    // 绑定深度纹理
    currentDepth->Bind(0);
    motionShader->SetInt("depthTexture", 0);
    
    // 绑定输出
    output->BindImage(0, 0, GL_WRITE_ONLY);
    
    // 分派
    uint32_t groupsX = (width + 15) / 16;
    uint32_t groupsY = (height + 15) / 16;
    motionShader->Dispatch(groupsX, groupsY, 1);
    
    ComputeShader::ImageAccessBarrier();
}

void MotionVectorCalculator::SetCurrentCamera(const glm::mat4& view, const glm::mat4& projection) {
    previousView = currentView;
    previousProjection = currentProjection;
    previousViewProjection = previousProjection * previousView;
    
    currentView = view;
    currentProjection = projection;
    currentViewProjectionInverse = glm::inverse(projection * view);
}

void MotionVectorCalculator::SetPreviousCamera(const glm::mat4& view, const glm::mat4& projection) {
    previousView = view;
    previousProjection = projection;
    previousViewProjection = previousProjection * previousView;
}

} // namespace HybridPBR
```

```cpp
// HybridDenoiser.cpp

#include "HybridDenoiser.h"
#include "utils/Logger.h"

namespace HybridPBR {

HybridDenoiser::HybridDenoiser() 
    : width(0), height(0) {
    temporalAccumulator = std::make_unique<TemporalAccumulator>();
    motionVectorCalc = std::make_unique<MotionVectorCalculator>();
    spatialDenoiser = std::make_unique<Denoiser>();
}

HybridDenoiser::~HybridDenoiser() {
}

bool HybridDenoiser::Initialize(uint32_t width, uint32_t height) {
    this->width = width;
    this->height = height;
    
    // 初始化子模块
    if (!temporalAccumulator->Initialize(width, height)) {
        LOG_ERROR("Failed to initialize temporal accumulator");
        return false;
    }
    
    if (!motionVectorCalc->Initialize(width, height)) {
        LOG_ERROR("Failed to initialize motion vector calculator");
        return false;
    }
    
    if (!spatialDenoiser->Initialize(width, height)) {
        LOG_ERROR("Failed to initialize spatial denoiser");
        return false;
    }
    
    // 创建中间纹理
    temporalResult = std::make_shared<Texture>();
    temporalResult->Create2D(width, height, GL_RGBA16F, GL_RGBA, GL_FLOAT);
    
    motionVectors = std::make_shared<Texture>();
    motionVectors->Create2D(width, height, GL_RG16F, GL_RG, GL_FLOAT);
    
    workTexture = std::make_shared<Texture>();
    workTexture->Create2D(width, height, GL_RGBA16F, GL_RGBA, GL_FLOAT);
    
    return true;
}

void HybridDenoiser::Process(
    const std::shared_ptr<Texture>& inputColor,
    const std::shared_ptr<Texture>& gbufferDepth,
    const std::shared_ptr<Texture>& gbufferNormal,
    std::shared_ptr<Texture>& outputColor) {
    
    std::shared_ptr<Texture> currentStage = inputColor;
    
    // 步骤1：计算运动向量
    if (config.enableTemporalAccumulation || config.enableTAA) {
        motionVectorCalc->Calculate(gbufferDepth, motionVectors);
    }
    
    // 步骤2：时域累积
    if (config.enableTemporalAccumulation) {
        temporalAccumulator->Accumulate(
            currentStage, motionVectors, 
            gbufferDepth, gbufferNormal,
            temporalResult
        );
        currentStage = temporalResult;
    }
    
    // 步骤3：空域降噪
    if (config.enableSpatialDenoising) {
        spatialDenoiser->Denoise(
            currentStage, workTexture,
            config.spatialStrength,
            gbufferNormal, gbufferDepth
        );
        currentStage = workTexture;
    }
    
    // 复制到输出
    if (currentStage != outputColor) {
        glCopyImageSubData(
            currentStage->GetID(), GL_TEXTURE_2D, 0, 0, 0, 0,
            outputColor->GetID(), GL_TEXTURE_2D, 0, 0, 0, 0,
            width, height, 1
        );
    }
}

void HybridDenoiser::BeginFrame(const glm::mat4& view, const glm::mat4& projection) {
    motionVectorCalc->SetCurrentCamera(view, projection);
    
    // 生成抖动
    uint32_t frameNumber = 0; // 从某处获取帧号
    glm::vec2 jitter = SobolSampler::Get2D(frameNumber % 256) - 0.5f;
    
    temporalAccumulator->SetJitter(jitter, glm::vec2(0.0f)); // 上一帧抖动需要保存
}

void HybridDenoiser::EndFrame() {
    temporalAccumulator->AdvanceFrame();
}

void HybridDenoiser::Resize(uint32_t width, uint32_t height) {
    if (width == this->width && height == this->height) return;
    
    this->width = width;
    this->height = height;
    
    temporalAccumulator->Resize(width, height);
    motionVectorCalc->Resize(width, height);
    spatialDenoiser->Resize(width, height);
    
    // 重新创建中间纹理
    temporalResult->Create2D(width, height, GL_RGBA16F, GL_RGBA, GL_FLOAT);
    motionVectors->Create2D(width, height, GL_RG16F, GL_RG, GL_FLOAT);
    workTexture->Create2D(width, height, GL_RGBA16F, GL_RGBA, GL_FLOAT);
}

} // namespace HybridPBR
```

## 7.3 着色器实现

### 7.3.1 运动向量着色器

```glsl
// motion_vectors.comp

#version 460 core
layout(local_size_x = 16, local_size_y = 16) in;

layout(binding = 0) uniform sampler2D depthTexture;
layout(binding = 0, rg16f) uniform image2D outputImage;

uniform mat4 currentViewProjectionInverse;
uniform mat4 previousViewProjection;
uniform int width;
uniform int height;

void main() {
    ivec2 coords = ivec2(gl_GlobalInvocationID.xy);
    if (coords.x >= width || coords.y >= height) return;
    
    // 读取深度
    float depth = texelFetch(depthTexture, coords, 0).r;
    
    // 重建世界位置
    vec2 uv = (vec2(coords) + 0.5) / vec2(width, height);
    vec2 ndc = uv * 2.0 - 1.0;
    vec4 clipPos = vec4(ndc, depth, 1.0);
    
    vec4 worldPos = currentViewProjectionInverse * clipPos;
    worldPos.xyz /= worldPos.w;
    
    // 投影到上一帧
    vec4 previousClipPos = previousViewProjection * worldPos;
    vec2 previousNDC = previousClipPos.xy / previousClipPos.w;
    vec2 previousUV = previousNDC * 0.5 + 0.5;
    
    // 计算运动向量（像素空间）
    vec2 motionVector = (previousUV - uv) * vec2(width, height);
    
    imageStore(outputImage, coords, vec4(motionVector, 0.0, 1.0));
}
```

### 7.3.2 时域累积着色器

```glsl
// temporal_accumulation.comp

#version 460 core
layout(local_size_x = 16, local_size_y = 16) in;

layout(binding = 0) uniform sampler2D currentColor;
layout(binding = 1) uniform sampler2D historyColor;
layout(binding = 2) uniform sampler2D motionVectors;
layout(binding = 3) uniform sampler2D currentDepth;
layout(binding = 4) uniform sampler2D historyDepth;
layout(binding = 5) uniform sampler2D currentNormal;
layout(binding = 6) uniform sampler2D historyNormal;

layout(binding = 0, rgba16f) uniform image2D outputImage;

uniform int width;
uniform int height;
uniform float blendFactor;
uniform vec2 currentJitter;
uniform vec2 previousJitter;

vec3 clipAABB(vec3 history, vec3 minColor, vec3 maxColor) {
    vec3 center = 0.5 * (maxColor + minColor);
    vec3 extents = 0.5 * (maxColor - minColor);
    vec3 displacement = history - center;
    vec3 absDisplacement = abs(displacement);
    vec3 ts = extents / (absDisplacement + 0.0001);
    float t = min(min(ts.x, ts.y), ts.z);
    t = clamp(t, 0.0, 1.0);
    return center + displacement * t;
}

void main() {
    ivec2 coords = ivec2(gl_GlobalInvocationID.xy);
    if (coords.x >= width || coords.y >= height) return;
    
    vec2 texelSize = 1.0 / vec2(width, height);
    vec2 uv = (vec2(coords) + 0.5) * texelSize;
    
    // 当前颜色
    vec3 current = texelFetch(currentColor, coords, 0).rgb;
    
    // 计算邻域范围
    vec3 minColor = current;
    vec3 maxColor = current;
    for (int y = -1; y <= 1; y++) {
        for (int x = -1; x <= 1; x++) {
            vec3 neighbor = texelFetch(currentColor, 
                clamp(coords + ivec2(x, y), ivec2(0), ivec2(width-1, height-1)), 0).rgb;
            minColor = min(minColor, neighbor);
            maxColor = max(maxColor, neighbor);
        }
    }
    
    // 读取运动向量
    vec2 motion = texelFetch(motionVectors, coords, 0).rg;
    
    // 补偿抖动
    motion += previousJitter;
    motion -= currentJitter;
    
    // 历史采样位置
    vec2 historyUV = uv + motion * texelSize;
    
    // 有效性检测
    bool validHistory = all(greaterThanEqual(historyUV, vec2(0.0))) &&
                        all(lessThan(historyUV, vec2(1.0)));
    
    // 深度检测
    float currentDepth = texelFetch(currentDepth, coords, 0).r;
    if (validHistory) {
        float historyDepth = texture(historyDepth, historyUV).r;
        if (abs(currentDepth - historyDepth) > 0.01) {
            validHistory = false;
        }
    }
    
    // 法线检测
    if (validHistory) {
        vec3 currentNormal = texelFetch(currentNormal, coords, 0).rgb;
        vec3 historyNormal = texture(historyNormal, historyUV).rgb;
        if (dot(currentNormal, historyNormal) < 0.9) {
            validHistory = false;
        }
    }
    
    // 混合
    vec3 finalColor;
    if (validHistory) {
        vec3 history = texture(historyColor, historyUV).rgb;
        history = clipAABB(history, minColor, maxColor);
        finalColor = mix(history, current, blendFactor);
    } else {
        finalColor = current;
    }
    
    imageStore(outputImage, coords, vec4(finalColor, 1.0));
}
```

```cpp
        }
        sceneDirty = false;
        
        // 新增：重置时域累积
        if (hybridDenoiser) {
            hybridDenoiser->Reset();
        }
    }
    
    // 生成光线
    GenerateRays();
    
    // 路径追踪
    TracePaths();
    
    // 复制到纹理
    CopyToTexture();
    
    // 新增：使用整合降噪器
    if (hybridDenoiser) {
        auto gbuffer = GetGBuffer(); // 需要实现
        hybridDenoiser->Process(
            outputTexture,
            gbuffer->GetTexture(GBufferTextureType::Position), // 或深度纹理
            gbuffer->GetTexture(GBufferTextureType::Normal),
            denoisedTexture
        );
        hybridDenoiser->EndFrame();
    } 
    // 保留原有降噪作为备选
    else if (config.denoiseEnabled) {
        DenoiseResult();
    }
    
    accumulatedFrames++;
    
    auto endTime = std::chrono::high_resolution_clock::now();
    lastRenderTime = std::chrono::duration<float, std::milli>(endTime - startTime).count();
}
```

## 7.6 性能优化建议

### 7.6.1 内存带宽优化

```
优化策略：
┌─────────────────────────────────────────────────────────────┐
│  1. 使用压缩纹理格式                                         │
│     - 运动向量：RG16F（2个16位浮点）                        │
│     - 颜色：RGBA16F（4个16位浮点）                          │
│     - 避免使用RGBA32F除非必要                               │
│                                                             │
│  2. 减少纹理读写                                             │
│     - 合并多个pass到单个着色器                              │
│     - 使用shared memory缓存邻域数据                         │
│                                                             │
│  3. 异步计算                                                 │
│     - 运动向量计算可以与前向渲染并行                        │
│     - 使用GL异步内存传输                                    │
└─────────────────────────────────────────────────────────────┘
```

### 7.6.2 计算优化

```
优化策略：
┌─────────────────────────────────────────────────────────────┐
│  1. 分离滤波                                                 │
│     - 将2D滤波分解为水平+垂直两遍                           │
│     - 从O(n²)降到O(2n)                                      │
│                                                             │
│  2. 早期退出                                                 │
│     - 检测无效像素提前返回                                  │
│     - 权重太小时跳过计算                                    │
│                                                             │
│  3. SIMD优化                                                 │
│     - 使用GLSL的向量操作                                    │
│     - 避免分支 divergence                                   │
└─────────────────────────────────────────────────────────────┘
```
